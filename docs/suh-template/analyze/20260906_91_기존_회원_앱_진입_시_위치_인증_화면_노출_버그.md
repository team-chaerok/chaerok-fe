class LocationVerificationFailed extends LocationVerificationOutcome {
  const LocationVerificationFailed(
    this.reason, {
    this.isLocationServiceEnabled = true,

  final LocationVerificationFailureReason reason;

  final bool isLocationServiceEnabled;
}

enum LocationVerificationFailureReason {
  permissionDenied,
  permissionPermanentlyDenied,
  locationUnavailable,
  regionVerificationFailed,
  placesFailed,
}

Future<void> _run() async {
  setState(() => _step = _Step.checking);
  final outcome = await LocationVerificationRunner.run();
  if (!mounted) return;
  switch (outcome) {
    case LocationVerified(:final result):

```dart
Future<LocationVerificationOutcome> _runLocationVerification() =>
    (widget.debugRunLocationVerification ?? LocationVerificationRunner.run)();

Future<void> _ensureLocationVerified() async {
  final cached = LocationVerificationResult.sessionCache;
  if (cached != null) {
    setState(() => _locationResult = cached);
    unawaited(_onLocationVerified(cached));
    return;
  }
  if (LocationVerificationResult.outOfServiceSessionCache) {
    setState(() => _isOutOfService = true);
    return;
  }

  // 기존 회원: 화면 전환 없이 조용히 위치 확인.
  final outcome = await _runLocationVerification();
  if (!mounted) return;
  if (outcome is! LocationVerificationFailed) {
    _applyOutcome(outcome);
    return;
  }

  // 권한 미허용·좌표/지역/관광지 조회 실패 → 온보딩 뷰 없이 에러 안내 화면으로 폴백.
  final fallback = await Navigator.of(context).push<LocationVerificationOutcome>(
    MaterialPageRoute(builder: (_) => LocationVerificationScreen(initialFailure: outcome)),
  );
  if (!mounted) return;
  _applyOutcome(fallback);
}

void _applyOutcome(LocationVerificationOutcome? outcome) {
  switch (outcome) {
    case LocationVerified(:final result):
      setState(() => _locationResult = result);
      unawaited(_onLocationVerified(result));
    case LocationOutOfService():
      setState(() => _isOutOfService = true);
    case LocationVerificationFailed():
    case null:
      // 폴백 화면을 뒤로가기로 닫음(null) 등: 게이트가 세팅했을 수 있는 캐시 재확인.
      if (LocationVerificationResult.outOfServiceSessionCache) {
        setState(() => _isOutOfService = true);
      }
  }
}
```

- `_runLocationVerification`의 tear-off 주의: `LocationVerificationRunner.run`은 named optional 파라미터가 있어 `Future Function()`에 직접 대입 불가 → `?? () => LocationVerificationRunner.run()` 형태로 래핑하거나 별도 top-level 함수로.

### 대안 및 미채택

| 대안 | 미채택 사유 |
|------|-------------|
| **폴백 화면 `autoStart: true`로 러너 재실행** | idle 뷰는 건너뛰지만 권한/좌표/API를 한 번 더 태움(R7). `initialFailure` 전달이 더 저렴하고 결정적 |
| **화면에 `silent` 모드 플래그** (위젯 안 붙이고 `_run()`만) | 위젯을 로직 컨테이너로 오용. 생명주기 의존 잔존 |
| **홈에 인라인 구현** | `_run()`과 절차 이원화 → 드리프트. mock/디버그 분기 중복 |
| **세션 캐시 디스크 영속화** | 원인(회원 구분 없는 push) 미해결. 캐시 만료·지역 이동 stale 신규 발생. 별도 이슈 후보 |
| **스플래시에서 선(先) 위치 확인** | 스플래시 지연 증가, 권한 다이얼로그가 앱 첫 화면 앞에. 이번 범위 밖(러너 추출로 향후 가능) |

### 성능 / 보안

- 성능: 정상 경로에서 화면 push/pop(전환 애니메이션) 및 idle 뷰의 카카오맵 네이티브 뷰 로드가 사라져 **진입 체감 개선**. 러너는 기존 `_run()`과 동일 호출량.
- 보안: 신규 권한/토큰 취급 없음. Kakao REST 키는 기존 `KakaoLocalApiService` 경유 그대로.
- 세션 캐시 미영속으로 앱 재실행마다 Kakao/백엔드 재호출은 잔존(이슈 범위 밖, 후속 트래킹 권장).
